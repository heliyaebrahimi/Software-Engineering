
flowchart TD
    A([شروع]) --> B[نمایش کالاها و قیمت‌ها]
    B --> C[انتخاب کالا توسط مشتری]
    C --> D{آیا کالا موجود است؟}

    D -- خیر --> E[نمایش پیام ناموجود بودن کالا]
    E --> Z([پایان])

    D -- بله --> F[دریافت وجه]
    F --> G{آیا پرداخت تأیید شده است؟}

    G -- خیر --> H[نمایش پیام خطای پرداخت]
    H --> I[بازگرداندن وجه قابل استرداد]
    I --> Z

    G -- بله --> J[آغاز فرایند تحویل کالا]
    J --> K{آیا تحویل موفق بوده است؟}

    K -- خیر --> L[آغاز بازپرداخت و ثبت خطا]
    L --> Z

    K -- بله --> M[به‌روزرسانی موجودی کالا]
    M --> N[بازگرداندن باقی‌مانده وجه]
    N --> O[نمایش پیام موفقیت]
    O --> Z
